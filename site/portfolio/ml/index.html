import torch

# Educational toy data. The browser plot
# runs equivalent math in JavaScript.
torch.manual_seed(7)
x = torch.linspace(-1, 1, 21).view(-1, 1)
y = 1 + 1.8 * x + 0.16 * torch.sin(
    torch.arange(21).float() * 5
).view(-1, 1)

model = torch.nn.Linear(1, 1)
optim = torch.optim.SGD(model.parameters(),
                        lr=0.12)
loss_fn = torch.nn.MSELoss()

for epoch in range(180):
    pred = model(x)
    loss = loss_fn(pred, y)
    optim.zero_grad()
    loss.backward()
    optim.step()
    if epoch % 30 == 0:
        print(epoch, round(loss.item(), 4))

print("weight", model.weight.item())
print("bias", model.bias.item())
